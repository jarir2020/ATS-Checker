// ── Helpers ───────────────────────────────────────────────────────
function secOn(array $d, string $k): bool { return ($d['sec_on'][$k] ?? '0') === '1'; }
function av(array $d, string $k): array   { return array_values($d[$k] ?? []); }
function e(string $s): string             { return htmlspecialchars($s, ENT_QUOTES); }
function project_link_icon(string $url): string {
    return '<a class="pr-link" href="'.e($url).'" target="_blank" rel="noopener noreferrer" aria-label="Open project link" title="Open project link"><span aria-hidden="true">↗</span></a>';
}

function build_classic(array $d): string {
    $lines = ($d['show_lines'] ?? '1') !== '0';
    $h = '<html><head><meta charset="UTF-8"><style>'.css_classic($lines).'</style></head><body>';
    $h .= header_classic($d);
    $h .= body_classic($d);
    return $h.'</body></html>';
}

function header_classic(array $d): string {
    $parts = contact_parts($d);
    $h  = '<div class="cv-name">'.e($d['cv_name']??'').'</div>';
    if (!empty($d['cv_role'])) $h .= '<div class="cv-role">'.e($d['cv_role']).'</div>';
    $h .= '<hr class="cv-hr">';
    if ($parts) $h .= '<div class="cv-ct">'.implode(' | ',$parts).'</div>';
    $h .= '<hr class="cv-hr">';
    return $h;
}

function body_classic(array $d): string {
    $h = '';
    if (secOn($d,'summary') && !empty($d['summary']))
        $h .= '<div class="sec"><div class="st">Summary</div><p>'.e($d['summary']).'</p></div>';
    $h .= skills_classic($d).exp_classic($d).edu_classic($d)
         .projects_classic($d).certs_classic($d).langs_classic($d).extra_classic($d);
    return $h;
}

function skills_classic(array $d): string {
    if (!secOn($d,'skills')) return '';
    $rows=''; foreach (av($d,'skill_label') as $i=>$l) {
        $l=trim($l); $v=trim(av($d,'skill_value')[$i]??'');
        if (!$l&&!$v) continue;
        $rows.='<div class="sk-row"><span class="sk-lbl">'.e($l).':</span> '.e($v).'</div>';
    }
    return $rows ? '<div class="sec"><div class="st">Skills</div>'.$rows.'</div>' : '';
}
function exp_classic(array $d): string {
    if (!secOn($d,'experience')) return '';
    $bl=''; foreach (av($d,'exp_company') as $i=>$co) {
        if (!trim($co)) continue;
        $loc=trim(av($d,'exp_location')[$i]??''); $ttl=trim(av($d,'exp_title')[$i]??'');
        $st=trim(av($d,'exp_start')[$i]??'');     $en=trim(av($d,'exp_end')[$i]??'');
        $rsp=trim(av($d,'exp_resp')[$i]??'');
        $bl.='<div class="exp-e"><div class="exp-co">'.e($co).($loc?' | '.e($loc):'').'</div>';
        $bl.='<div>'.e($ttl).($st||$en?' | '.e($st).' – '.e($en):'').'</div>';
        if ($rsp) { $bl.='<div class="r-lbl">Responsibilities:</div>';
            foreach (array_filter(array_map('trim',explode("\n",$rsp))) as $x=>$ln)
                $bl.='<div class="r-itm">'.($x+1).'. '.e($ln).'</div>'; }
        $bl.='</div>';
    }
    return $bl ? '<div class="sec"><div class="st">Experience</div>'.$bl.'</div>' : '';
}
function edu_classic(array $d): string {
    if (!secOn($d,'education')) return '';
    $bl=''; foreach (av($d,'edu_uni') as $i=>$u) {
        if (!trim($u)) continue;
        $loc=trim(av($d,'edu_location')[$i]??''); $dg=trim(av($d,'edu_degree')[$i]??'');
        $yr=trim(av($d,'edu_year')[$i]??'');      $dc=trim(av($d,'edu_desc')[$i]??'');
        $bl.='<div class="edu-e"><div class="edu-uni">'.e($u).($loc?' | '.e($loc):'').'</div>';
        $bl.='<div>'.e($dg).($yr?' | '.e($yr):'').'</div>';
        if ($dc) $bl.='<div class="edu-dsc">'.e($dc).'</div>';
        $bl.='</div>';
    }
    return $bl ? '<div class="sec"><div class="st">Education</div>'.$bl.'</div>' : '';
}
function projects_classic(array $d): string {
    if (!secOn($d,'projects')) return '';
    $bl=''; foreach (av($d,'proj_name') as $i=>$n) {
        if (!trim($n)) continue;
        $tc=trim(av($d,'proj_tech')[$i]??''); $dc=trim(av($d,'proj_desc')[$i]??''); $ur=trim(av($d,'proj_url')[$i]??'');
        $bl.='<div class="pr-e"><div class="pr-hd">'.e($n).($tc?' <span class="pr-tch">— '.e($tc).'</span>':'').($ur?' '.project_link_icon($ur):'').'</div>';
        if ($dc) $bl.='<div class="pr-dsc">'.e($dc).'</div>';
        $bl.='</div>';
    }
    return $bl ? '<div class="sec"><div class="st">Projects</div>'.$bl.'</div>' : '';
}
function certs_classic(array $d): string {
    if (!secOn($d,'certifications')) return '';
    $bl=''; foreach (av($d,'cert_name') as $i=>$n) {
        if (!trim($n)) continue;
        $is=trim(av($d,'cert_issuer')[$i]??''); $dt=trim(av($d,'cert_date')[$i]??''); $id=trim(av($d,'cert_id')[$i]??'');
        $bl.='<div class="ct-e"><div class="ct-nm">'.e($n).($is?' — '.e($is):'').($dt?' | '.e($dt):'').'</div>';
        if ($id) $bl.='<div class="ct-id">Credential ID: '.e($id).'</div>';
        $bl.='</div>';
    }
    return $bl ? '<div class="sec"><div class="st">Certifications</div>'.$bl.'</div>' : '';
}
function langs_classic(array $d): string {
    if (!secOn($d,'languages')) return '';
    $rows=''; foreach (av($d,'lang_name') as $i=>$n) {
        if (!trim($n)) continue;
        $lv = trim(av($d,'lang_level')[$i]??'');
        $rows.='<div class="lg-row"><span class="lg-nm">'.e($n).'</span>'.($lv ? ' — '.e($lv) : '').'</div>';
    }
    return $rows ? '<div class="sec"><div class="st">Languages</div>'.$rows.'</div>' : '';
}
function extra_classic(array $d): string {
    if (!secOn($d,'extra')) return '';
    $h=''; foreach (av($d,'extra_title') as $i=>$t) {
        if (!trim($t)) continue;
        $h.='<div class="ex-sec"><div class="st">'.e($t).'</div><div>'.e(av($d,'extra_content')[$i]??'').'</div></div>';
    }
    return $h;
}

function build_modern(array $d): string {
    $lines = ($d['show_lines'] ?? '1') !== '0';
    $h = '<html><head><meta charset="UTF-8"><style>'.css_modern($lines).'</style></head><body>';
    $h .= '<table class="cv-tbl"><tr>';
    // Sidebar
    $h .= '<td class="sb">';
    $h .= '<div class="sb-name">'.e($d['cv_name']??'').'</div>';
    if (!empty($d['cv_role'])) $h .= '<div class="sb-role">'.e($d['cv_role']).'</div>';
    // Contact
    $cp = contact_parts($d, true);
    if ($cp) { $h.='<div class="sb-st">Contact</div>'; foreach ($cp as $p) $h.='<div class="sb-ct">'.$p.'</div>'; }
    // Skills
    if (secOn($d,'skills')) {
        $rows=''; foreach (av($d,'skill_label') as $i=>$l) {
            $l=trim($l); $v=trim(av($d,'skill_value')[$i]??'');
            if (!$l&&!$v) continue;
            $rows.='<div class="sb-sk"><span class="sb-lbl">'.e($l).'</span><br>'.e($v).'</div>';
        }
        if ($rows) $h.='<div class="sb-st">Skills</div>'.$rows;
    }
    // Languages
    if (secOn($d,'languages')) {
        $rows=''; foreach (av($d,'lang_name') as $i=>$n) {
            if (!trim($n)) continue;
            $lv=trim(av($d,'lang_level')[$i]??'');
            $rows.='<div class="sb-lg"><span class="sb-lg-nm">'.e($n).'</span>'.($lv?' — '.e($lv):'').'</div>';
        }
        if ($rows) $h.='<div class="sb-st">Languages</div>'.$rows;
    }
    $h .= '</td>';
    // Main
    $h .= '<td class="mn">';
    if (secOn($d,'summary')&&!empty($d['summary']))
        $h.='<div class="mn-st">Summary</div><p>'.e($d['summary']).'</p>';
    // Experience (re-use classic builders with modern class st→mn-st)
    $h.=str_replace('"st"','"mn-st"', exp_classic($d).edu_classic($d).projects_classic($d).certs_classic($d).extra_classic($d));
    $h .= '</td></tr></table>';
    return $h.'</body></html>';
}

// ─────────────────────────────────────────────────────────────────
// Shared helpers
// ─────────────────────────────────────────────────────────────────
function contact_parts(array $d, bool $separate=false): array {
    $p=[];
    if (!empty($d['phone']))    $p[] = e($d['phone']);
    if (!empty($d['email']))    $p[] = '<a href="mailto:'.e($d['email']).'">'.e($d['email']).'</a>';
    if (!empty($d['location'])) $p[] = e($d['location']);
    if (!empty($d['linkedin'])) $p[] = '<a href="'.e($d['linkedin']).'">Linkedin</a>';
    if (!empty($d['github']))   $p[] = '<a href="'.e($d['github']).'">Github</a>';
    return $p;
}

function buildCvHtml(array $d): string {
    switch ($d['cv_template'] ?? 'classic') {
        case 'modern':       return build_modern($d);
        case 'professional': return build_professional($d);
        case 'minimal':      return build_minimal($d);
        case 'executive':    return str_replace(['#1e3a5f','#a8c8e8','#cde'], ['#1f2937','#cbd5e1','#e5e7eb'], build_professional($d));
        default:             return build_classic($d);
    }
}

// ── Draft AJAX ────────────────────────────────────────────────────
if (isset($_GET['action'])) {
    header('Content-Type: application/json');
    $db = getDb();
    if ($_GET['action'] === 'list_drafts') {
        echo json_encode($db->query("SELECT id, draft_name, updated_at FROM draft_cvs ORDER BY updated_at DESC")->fetchAll());
    } elseif ($_GET['action'] === 'get_draft' && isset($_GET['id'])) {
        $st = $db->prepare("SELECT cv_data FROM draft_cvs WHERE id=?");
        $st->execute([(int)$_GET['id']]);
        $row = $st->fetch();
        echo $row ? $row['cv_data'] : 'null';
    }
    exit;
}
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $act = $_POST['action'] ?? '';
    if ($act === 'save_draft') {
        header('Content-Type: application/json');
        $name = trim($_POST['draft_name'] ?? '');
        $data = $_POST['cv_data'] ?? '';
        $id   = (int)($_POST['draft_id'] ?? 0);
        if ($name && $data) {
            $db = getDb();
            if ($id) {
                $db->prepare("UPDATE draft_cvs SET draft_name=?,cv_data=?,updated_at=NOW() WHERE id=?")->execute([$name,$data,$id]);
            } else {
                $db->prepare("INSERT INTO draft_cvs (draft_name,cv_data) VALUES (?,?)")->execute([$name,$data]);
                $id = (int)$db->lastInsertId();
            }
            echo json_encode(['ok'=>true,'id'=>$id]);
        } else {
            echo json_encode(['ok'=>false,'msg'=>'Missing data']);
        }
        exit;
    }
    if ($act === 'delete_draft') {
        header('Content-Type: application/json');
        $id = (int)($_POST['id'] ?? 0);
        if ($id) getDb()->prepare("DELETE FROM draft_cvs WHERE id=?")->execute([$id]);
        echo json_encode(['ok'=>true]);
        exit;
    }
}

function stStyle(tmpl, lines) {
    const b = lines ? {
        classic:     'border-bottom:1px solid #aaa;',
        professional:'border-bottom:2px solid #1e3a5f;',
        minimal:     'border-bottom:1px dotted #aaa;',
        modern_sb:   'border-bottom:1px solid #3d5166;',
        modern_mn:   'border-left:3px solid #2c3e50;',
    } : {};
    const s = {
        classic:     `font-size:12px;font-weight:bold;${b.classic||''}padding-bottom:2px;margin-bottom:6px`,
        professional:`font-size:12px;font-weight:bold;color:#1e3a5f;${b.professional||''}padding-bottom:2px;margin-bottom:6px`,

let _draftId = 0;

function openSaveDraft() {
    document.getElementById('draftModal').style.display = 'flex';

function closeDraftModal() {
    document.getElementById('draftModal').style.display = 'none';

function newDraft() {
    _draftId = 0;
    document.getElementById('draftBadge').style.display = 'none';

function saveDraft() {
    const name = document.getElementById('draftModalName').value.trim();
    if (!name) { document.getElementById('draftModalName').focus(); return; }
    const fd = new FormData(document.getElementById('cvForm'));
    const obj = {};
    for (const [k, v] of fd.entries()) {
        if (k === 'action') continue;
        if (k.endsWith('[]')) {
            const key = k.slice(0,-2);
            if (!obj[key]) obj[key] = [];
            obj[key].push(v);
        } else if (/^\w+\[\w+\]$/.test(k)) {
            const [outer, rest] = k.split('[');
            const inner = rest.replace(']','');
            if (!obj[outer]) obj[outer] = {};
            obj[outer][inner] = v;
        } else {
            obj[k] = v;
        }
    }
    const body = new FormData();
    body.append('action','save_draft');
    body.append('draft_name', name);
    body.append('cv_data', JSON.stringify(obj));
    if (_draftId) body.append('draft_id', _draftId);
    fetch('', {method:'POST', body})
        .then(r => r.json())
        .then(r => {
            if (!r.ok) return;
            _draftId = r.id;
            closeDraftModal();
            loadDraftList();
            const b = document.getElementById('draftBadge');
            b.textContent = '✓ Draft: ' + name;
            b.style.display = 'block';
        });
}

function loadDraftList() {
    const el = document.getElementById('draftList');
    fetch('?action=list_drafts')
        .then(r => r.json())
        .then(rows => {
            if (!rows.length) {
                el.innerHTML = '<div style="color:#bbb;font-size:11px;padding:4px 0">No saved drafts.</div>';
                return;
            }
            el.innerHTML = rows.map(r => `
                <div style="display:flex;align-items:center;gap:6px;padding:5px 0;border-bottom:1px solid #f5f5f5">
                    <div style="flex:1;min-width:0">
                        <div style="font-weight:600;font-size:12px;color:#333;white-space:nowrap;overflow:hidden;text-overflow:ellipsis">${esc(r.draft_name)}</div>
                        <div style="font-size:10px;color:#bbb">${r.updated_at}</div>
                    </div>
                    <button data-id="${r.id}" data-name="${esc(r.draft_name)}"
                        onclick="loadDraft(+this.dataset.id,this.dataset.name)"
                        style="flex-shrink:0;font-size:11px;padding:2px 7px;border:1px solid #667eea;color:#667eea;background:none;border-radius:4px;cursor:pointer">Load</button>
                    <button data-id="${r.id}" onclick="delDraft(+this.dataset.id,this)"
                        style="flex-shrink:0;font-size:11px;padding:2px 7px;border:1px solid #e74c3c;color:#e74c3c;background:none;border-radius:4px;cursor:pointer">✕</button>
                </div>`).join('');
        });
}

function loadDraft(id, name) {
    fetch('?action=get_draft&id=' + id)
        .then(r => r.json())
        .then(d => {
            if (!d) return;
            fillForm(d);
            _draftId = id;
            document.getElementById('draftModalName').value = name;
            const b = document.getElementById('draftBadge');
            b.textContent = '✓ Draft: ' + name;
            b.style.display = 'block';
        });
}

function delDraft(id, btn) {
    if (!confirm('Delete this draft?')) return;
    const body = new FormData();
    body.append('action','delete_draft');
    body.append('id', id);
    fetch('', {method:'POST', body})
        .then(r => r.json())
        .then(() => {
            if (_draftId === id) newDraft();
            loadDraftList();
        });
}
